cmake_minimum_required(VERSION 3.13)


set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Release>:Release>")
set(CMAKE_VS_WINDOWS_TARGET_PLATFORM_MIN_VERSION "10.0.18362.0")

project(Mimic_Engine)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ENABLE_CLANG_TIDY "Enable clang-tidy static analysis" OFF)

if(ENABLE_CLANG_TIDY AND CMAKE_BUILD_TYPE STREQUAL "Debug")
	set(CMAKE_CXX_CLANG_TIDY
	clang-tidy;
	-p=${CMAKE_BINARY_DIR};
	)
endif()

file(GLOB_RECURSE MY_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_executable(Mimic_Engine WIN32 ${MY_SOURCES})

target_compile_features(Mimic_Engine PRIVATE cxx_std_17)
target_compile_definitions(${CMAKE_PROJECT_NAME} PUBLIC RESOURCES_PATH="${CMAKE_CURRENT_SOURCE_DIR}/resources/"
	PUBLIC SHADERS_PATH="${CMAKE_CURRENT_SOURCE_DIR}/shaders/"
	CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/config/") # Comment this if you compile a release version and uncomment the next line
#target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC RESOURCES_PATH="./resources/" SHADERS_PATH="./shaders/" CONFIG_PATH="./config/") # Uncomment this line to setup the PATH macro to the final directory when you share the project

# Third party sources directory
set(IMGUI_DIR ${CMAKE_SOURCE_DIR}/external/imgui)
set(IMPLOT_DIR ${CMAKE_SOURCE_DIR}/external/implot)

# Third party source files
set(IMGUI_SOURCES
    ${IMGUI_DIR}/imgui.cpp
    ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp
    ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/imgui_demo.cpp

    ${IMGUI_DIR}/backends/imgui_impl_win32.cpp
    ${IMGUI_DIR}/backends/imgui_impl_dx12.cpp
)
set(IMPLOT_SOURCES
	${IMPLOT_DIR}/implot.cpp
	${IMPLOT_DIR}/implot_items.cpp
)

target_include_directories(Mimic_Engine BEFORE PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include/" ${IMGUI_DIR} ${IMPLOT_DIR})

#Exclude thrid party sources from clang-tidy
add_library(no_static_analysis_code OBJECT ${IMGUI_SOURCES} ${IMPLOT_SOURCES})
if(ENABLE_CLANG_TIDY AND CMAKE_BUILD_TYPE STREQUAL "Debug")
	set_target_properties(no_static_analysis_code PROPERTIES CXX_CLANG_TIDY "")
endif()
target_include_directories(no_static_analysis_code SYSTEM PUBLIC ${IMGUI_DIR})
target_sources(Mimic_Engine PRIVATE $<TARGET_OBJECTS:no_static_analysis_code>)

#Link libraries
target_link_libraries(Mimic_Engine PRIVATE d3d12 dxgi dxguid dxcompiler.lib ${CMAKE_CURRENT_SOURCE_DIR}/thirdParty/meshoptimizer/lib/meshoptimizer.lib $<$<CONFIG:Debug>:${CMAKE_CURRENT_SOURCE_DIR}/thirdParty/DirectXTex/lib/DirectXTex_debug.lib>
    $<$<CONFIG:Release>:${CMAKE_CURRENT_SOURCE_DIR}/thirdParty/DirectXTex/lib/DirectXTex.lib>)
#Copy dll files to .exe directory
add_custom_command(TARGET Mimic_Engine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/dxc/bin/dxcompiler.dll $<TARGET_FILE_DIR:Mimic_Engine>)
